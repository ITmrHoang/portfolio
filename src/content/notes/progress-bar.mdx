Cách dùng trực tiếp trên HTML bằng js thường thấy hay dùng nhưng để làm progress bar bạn chỉ cần các thuộc tính css là đủ

```html
  const readProgress = () => {
      let scrollTop = document.documentElement.scrollTop || document.body.scrollTop;
      let clientHeight = document.documentElement.clientHeight;
      let scrollHeight = document.documentElement.scrollHeight;
      let progress = Math.min((scrollTop / (scrollHeight - clientHeight)) * 100, 100);
      document.querySelector('.progress-bar').style.width = progress + '%';
  };
  window.addEventListener('scroll', readProgress);
```

```js
const article = document.querySelector("article");
const progress = document.querySelector(".progress");

const observer = new IntersectionObserver(
  ([entry]) => {
    progress.style.transform =
      `scaleX(${entry.intersectionRatio})`;
  },
  {
    threshold: Array.from(
      { length: 101 },
      (_, i) => i / 100
    )
  }
);

observer.observe(article);
```
